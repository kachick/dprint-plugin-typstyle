== formats only touched statement ==
[|#set text(  size: 10pt  )|]
#set text(  fill: red  )

[expect]
[|#set text(size: 10pt)|]
#set text(  fill: red  )

== formats second statement while keeping first statement untouched ==
#set text(  size: 10pt  )
[|#set text(  fill: red  )|]

[expect]
#set text(  size: 10pt  )
[|#set text(fill: red)|]

== formats inside code block ==
#{
  [|let x   =   1|]
  let y   =   2
}

[expect]
#{
  [|let x = 1|]
  let y   =   2
}
